// A small CDP driver for one Ritemark dev instance: node cdp.mjs <port> <action> [args...]
//   eval <expr>                 evaluate in the workbench page, print the JSON value
//   cmd <palette label>         run a command from the palette (picks the row with that label)
//   shot <file>                 screenshot the window
//   places                      the places row's items and which is checked, the left column's title
//   wv <title part> <expr>      evaluate inside the webview whose active frame's document matches (title or body text)
//   key <code> [mods]           press a key in the page (mods: 4 = Cmd, 8 = Shift)
//   type <text>                 insert text where the focus is
import { writeFileSync } from 'node:fs';

const [port, action, ...rest] = process.argv.slice(2);
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

async function session(match) {
	const list = await (await fetch(`http://127.0.0.1:${port}/json/list`)).json();
	const target = list.find(match);
	if (!target) {
		throw new Error('no such target');
	}
	const socket = new WebSocket(target.webSocketDebuggerUrl);
	await new Promise((resolve, reject) => { socket.addEventListener('open', resolve, { once: true }); socket.addEventListener('error', reject, { once: true }); });
	let id = 0;
	const pending = new Map();
	socket.addEventListener('message', event => {
		const message = JSON.parse(event.data);
		if (message.id && pending.has(message.id)) {
			const { resolve, reject } = pending.get(message.id);
			pending.delete(message.id);
			message.error ? reject(new Error(JSON.stringify(message.error))) : resolve(message.result);
		}
	});
	const send = (method, params = {}) => new Promise((resolve, reject) => { const n = ++id; pending.set(n, { resolve, reject }); socket.send(JSON.stringify({ id: n, method, params })); });
	const evaluate = async expression => {
		const { result, exceptionDetails } = await send('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true });
		if (exceptionDetails) {
			throw new Error(JSON.stringify(exceptionDetails).slice(0, 800));
		}
		return result.value;
	};
	return { send, evaluate, close: () => socket.close(), target };
}

const workbench = () => session(t => t.type === 'page' && t.url.includes('workbench'));

async function key(page, code, modifiers = 0, keyName) {
	const name = keyName ?? (code.startsWith('Key') ? code.slice(3).toLowerCase() : code);
	const vk = { Escape: 27, Enter: 13, Tab: 9, Backspace: 8, ArrowDown: 40, ArrowUp: 38, End: 35 }[code] ?? name.toUpperCase().charCodeAt(0);
	await page.send('Input.dispatchKeyEvent', { type: 'rawKeyDown', modifiers, code, key: name, windowsVirtualKeyCode: vk });
	await page.send('Input.dispatchKeyEvent', { type: 'keyUp', modifiers, code, key: name, windowsVirtualKeyCode: vk });
}

async function clickAt(page, x, y) {
	await page.send('Input.dispatchMouseEvent', { type: 'mouseMoved', x, y });
	await page.send('Input.dispatchMouseEvent', { type: 'mousePressed', x, y, button: 'left', clickCount: 1 });
	await page.send('Input.dispatchMouseEvent', { type: 'mouseReleased', x, y, button: 'left', clickCount: 1 });
}

const QUICK_FOCUSED = `(() => { const e = document.activeElement; return !!(e && e.closest('.quick-input-widget') && e.tagName === 'INPUT'); })()`;

async function command(page, label) {
	await key(page, 'Escape');
	await key(page, 'KeyP', 4 | 8, 'p');
	for (let i = 0; i < 20 && !(await page.evaluate(QUICK_FOCUSED)); i++) {
		await sleep(150);
	}
	if (!(await page.evaluate(QUICK_FOCUSED))) {
		const bar = await page.evaluate(`(() => { const r = document.querySelector('.part.statusbar').getBoundingClientRect(); return [r.x + 300, r.y + r.height / 2]; })()`);
		await clickAt(page, ...bar);
		await key(page, 'KeyP', 4 | 8, 'p');
		await sleep(600);
	}
	if (!(await page.evaluate(QUICK_FOCUSED))) {
		throw new Error('the palette did not take the focus');
	}
	await page.send('Input.insertText', { text: label });
	const row = `[...document.querySelectorAll('.quick-input-widget .quick-input-list .monaco-list-row')].find(row => { const a = row.getAttribute('aria-label') || ''; return a === ${JSON.stringify(label)} || a.startsWith(${JSON.stringify(label + ',')}) || a.startsWith(${JSON.stringify(label + ' ')}); })`;
	for (let i = 0; i < 25 && !(await page.evaluate(`!!(${row})`)); i++) {
		await sleep(200);
	}
	const centre = await page.evaluate(`(() => { const e = ${row}; if (!e) return null; const r = e.getBoundingClientRect(); return [r.x + r.width / 2, r.y + r.height / 2]; })()`);
	if (!centre) {
		throw new Error(`no palette row "${label}"`);
	}
	await clickAt(page, ...centre);
}

const PLACES = `(() => {
	const items = [...document.querySelectorAll('.part.sidebar .composite-bar .action-item, .part.sidebar .pane-composite-part .composite-bar .action-item')];
	const seen = new Set();
	const row = items.filter(e => { if (seen.has(e)) return false; seen.add(e); const r = e.getBoundingClientRect(); return r.width > 0; }).map(e => {
		const a = e.querySelector('.action-label');
		const label = (a && (a.getAttribute('aria-label') || a.getAttribute('title'))) || e.getAttribute('aria-label') || '';
		return (e.classList.contains('checked') ? '*' : '') + label.split(/[\\n(]/)[0].trim();
	});
	const title = document.querySelector('.part.sidebar .title-label h2, .part.sidebar .composite.title .title-label')?.textContent ?? null;
	const head = [...document.querySelectorAll('.part.sidebar .ritemark-project-head *')].filter(e => e.children.length === 0 && e.textContent.trim()).map(e => e.textContent.trim()).slice(0, 8);
	const content = document.querySelector('.part.sidebar .content')?.getBoundingClientRect();
	const panes = [...document.querySelectorAll('.part.sidebar .pane-header, .part.sidebar .split-view-view')].length;
	return { row, title, head, panes, sidebarVisible: !!document.querySelector('.part.sidebar') && document.querySelector('.part.sidebar').getBoundingClientRect().width > 0 };
})()`;

const page = await workbench();
try {
	if (action === 'eval') {
		console.log(JSON.stringify(await page.evaluate(rest.join(' ')), null, 1));
	} else if (action === 'cmd') {
		await command(page, rest.join(' '));
		console.log('ran', rest.join(' '));
	} else if (action === 'shot') {
		const shot = await page.send('Page.captureScreenshot', { format: 'png' });
		writeFileSync(rest[0], Buffer.from(shot.data, 'base64'));
		console.log('saved', rest[0]);
	} else if (action === 'places') {
		console.log(JSON.stringify(await page.evaluate(PLACES)));
	} else if (action === 'click') {
		await clickAt(page, Number(rest[0]), Number(rest[1]));
	} else if (action === 'key') {
		await key(page, rest[0], Number(rest[1] ?? 0));
	} else if (action === 'type') {
		await page.send('Input.insertText', { text: rest.join(' ') });
	} else if (action === 'wv') {
		page.close();
		const list = await (await fetch(`http://127.0.0.1:${port}/json/list`)).json();
		const frames = list.filter(t => t.type === 'iframe' && t.url.startsWith('vscode-webview://'));
		const want = rest[0];
		const expr = rest.slice(1).join(' ');
		let done = false;
		for (const frame of frames) {
			const s = await session(t => t.id === frame.id);
			try {
				const ok = await s.evaluate(`(() => { const af = document.getElementById('active-frame'); const d = af && af.contentDocument; if (!d) return false; return (d.title || '').includes(${JSON.stringify(want)}) || (d.body && d.body.innerText.includes(${JSON.stringify(want)})); })()`);
				if (ok) {
					const value = await s.evaluate(`(() => { const af = document.getElementById('active-frame'); const d = af.contentDocument; const w = af.contentWindow; return (${expr}); })()`);
					console.log(JSON.stringify(value, null, 1));
					done = true;
				}
			} finally {
				s.close();
			}
			if (done) {
				break;
			}
		}
		if (!done) {
			console.log('no webview matched', want);
		}
	}
} finally {
	page.close();
}
